cmake_minimum_required(VERSION 3.20)
project(OrderedPatternsCore VERSION 2.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_STANDARD 17)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
option(PATTERN_METAL "Build the runtime-compiled Metal backend on macOS" ${APPLE})
find_package(Threads REQUIRED)
add_library(pattern_core STATIC src/PatternCore.cpp src/RasterCore.cpp)
target_include_directories(pattern_core PUBLIC src)
if(APPLE AND PATTERN_METAL)
    enable_language(OBJCXX)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/RasterMetalSource.h"
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/embed_metal.py"
                "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}"
        DEPENDS src/RasterMetal.metal src/RasterShared.h scripts/embed_metal.py VERBATIM)
    target_sources(pattern_core PRIVATE src/RasterMetal.mm "${CMAKE_CURRENT_BINARY_DIR}/RasterMetalSource.h")
    target_include_directories(pattern_core PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    target_compile_options(pattern_core PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
    target_link_libraries(pattern_core PUBLIC "-framework Foundation" "-framework Metal")
else()
    target_sources(pattern_core PRIVATE src/RasterGPUStub.cpp)
endif()
enable_testing()
add_executable(core_test tests/core_test.cpp)
target_link_libraries(core_test PRIVATE pattern_core Threads::Threads)
add_test(NAME core COMMAND core_test)
add_executable(raster_test tests/raster_test.cpp)
target_link_libraries(raster_test PRIVATE pattern_core Threads::Threads)
if(APPLE AND PATTERN_METAL)
    add_test(NAME raster COMMAND raster_test --metal)
else()
    add_test(NAME raster COMMAND raster_test)
endif()
# Tests use assertions even when a Release configuration defines NDEBUG.
foreach(test_target core_test raster_test)
    if(MSVC)
        target_compile_options(${test_target} PRIVATE /UNDEBUG /utf-8)
    else()
        target_compile_options(${test_target} PRIVATE -UNDEBUG)
    endif()
endforeach()
